JWT_ALGORITHM = "HS256"


def jwt_secret():
    return os.environ["JWT_SECRET"]


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


def verify_password(plain: str, hashed: str) -> bool:
    return bcrypt.checkpw(plain.encode("utf-8"), hashed.encode("utf-8"))


def create_access_token(user_id: str, email: str) -> str:
    payload = {
        "sub": user_id,
        "email": email,
        "exp": datetime.now(timezone.utc) + timedelta(minutes=30),
        "type": "access",
    }
    return jwt.encode(payload, jwt_secret(), algorithm=JWT_ALGORITHM)


def create_refresh_token(user_id: str) -> str:
    payload = {
        "sub": user_id,
        "exp": datetime.now(timezone.utc) + timedelta(days=7),
        "type": "refresh",
    }
    return jwt.encode(payload, jwt_secret(), algorithm=JWT_ALGORITHM)


def set_auth_cookies(response: Response, user_id: str, email: str):
    response.set_cookie(
        key="access_token", value=create_access_token(user_id, email),
        httponly=True, secure=True, samesite="none", max_age=1800, path="/",
    )
    response.set_cookie(
        key="refresh_token", value=create_refresh_token(user_id),
        httponly=True, secure=True, samesite="none", max_age=604800, path="/",
    )


def public_user(u: dict) -> dict:
    return {
        "id": str(u["_id"]),
        "email": u["email"],
        "name": u.get("name", ""),
        "role": u.get("role", "parent"),
    }


async def get_current_user(request: Request) -> dict:
    token = request.cookies.get("access_token")
    if not token:
        auth_header = request.headers.get("Authorization", "")
        if auth_header.startswith("Bearer "):
            token = auth_header[7:]
    if not token:
        raise HTTPException(status_code=401, detail="Not authenticated")
    try:
        payload = jwt.decode(token, jwt_secret(), algorithms=[JWT_ALGORITHM])
        if payload.get("type") != "access":
            raise HTTPException(status_code=401, detail="Invalid token type")
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token expired")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="Invalid token")
    user = await db.users.find_one({"_id": payload["sub"]})
    if not user:
        raise HTTPException(status_code=401, detail="User not found")
    return user


class RegisterIn(BaseModel):
    name: str
    email: EmailStr
    password: str
    role: str = "parent"


class LoginIn(BaseModel):
    email: EmailStr
    password: str


@api.post("/auth/register")
async def register(body: RegisterIn, response: Response):
    email = body.email.lower()
    if await db.users.find_one({"email": email}):
        raise HTTPException(status_code=400, detail="An account with this email already exists")
    if len(body.password) < 8:
        raise HTTPException(status_code=400, detail="Password must be at least 8 characters")
    uid = str(uuid.uuid4())
    user = {
        "_id": uid,
        "email": email,
        "name": body.name.strip(),
        "role": body.role if body.role in ("parent", "professional") else "parent",
        "password_hash": hash_password(body.password),
        "created_at": datetime.now(timezone.utc).isoformat(),
    }
    await db.users.insert_one(user)
    set_auth_cookies(response, uid, email)
    return public_user(user)


@api.post("/auth/login")
async def login(body: LoginIn, request: Request, response: Response):
    email = body.email.lower()
    identifier = f"{request.client.host if request.client else 'unknown'}:{email}"
    attempt = await db.login_attempts.find_one({"identifier": identifier})
    if attempt and attempt.get("count", 0) >= 5:
        locked_until = attempt.get("locked_until")
        if locked_until and datetime.fromisoformat(locked_until) > datetime.now(timezone.utc):
            raise HTTPException(status_code=429, detail="Too many failed attempts. Try again in 15 minutes.")
    user = await db.users.find_one({"email": email})
    if not user or not verify_password(body.password, user["password_hash"]):
        await db.login_attempts.update_one(
            {"identifier": identifier},
            {
                "$inc": {"count": 1},
                "$set": {"locked_until": (datetime.now(timezone.utc) + timedelta(minutes=15)).isoformat()},
            },
            upsert=True,
        )
        raise HTTPException(status_code=401, detail="Invalid email or password")
    await db.login_attempts.delete_one({"identifier": identifier})
    set_auth_cookies(response, user["_id"], email)
    return public_user(user)


@api.post("/auth/logout")
async def logout(response: Response):
    response.delete_cookie("access_token", path="/")
    response.delete_cookie("refresh_token", path="/")
    return {"ok": True}


@api.get("/auth/me")
async def me(user=Depends(get_current_user)):
    return public_user(user)


@api.post("/auth/refresh")
async def refresh(request: Request, response: Response):
    token = request.cookies.get("refresh_token")
    if not token:
        raise HTTPException(status_code=401, detail="No refresh token")
    try:
        payload = jwt.decode(token, jwt_secret(), algorithms=[JWT_ALGORITHM])
        if payload.get("type") != "refresh":
            raise HTTPException(status_code=401, detail="Invalid token type")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="Invalid refresh token")
    user = await db.users.find_one({"_id": payload["sub"]})
    if not user:
        raise HTTPException(status_code=401, detail="User not found")
    response.set_cookie(
        key="access_token", value=create_access_token(user["_id"], user["email"]),
        httponly=True, secure=True, samesite="none", max_age=1800, path="/",
    )
    return {"ok": True}


@api.get("/assessment-types")
async def list_types():
    return [
        {
            "key": k,
            "title": ASSESSMENT_TYPES[k]["title"],
            "description": ASSESSMENT_TYPES[k]["description"],
            "topic_count": len(ASSESSMENT_TYPES[k]["topics"]),
        }
        for k in ORDERED_TYPE_KEYS
    ]


@api.get("/assessment-types/{key}")
async def get_type(key: str):
    cfg = ASSESSMENT_TYPES.get(key)
    if not cfg:
        raise HTTPException(status_code=404, detail="Unknown assessment type")
    return {
        "key": cfg["key"],
        "title": cfg["title"],
        "description": cfg["description"],
        "central_question": cfg["central_question"],
        "scale": SCALE,
        "shared_goal_options": cfg["shared_goal_options"],
        "topics": [{"key": t["key"], "label": t["label"], "question": t["question"]} for t in cfg["topics"]],
    }


def case_public(c: dict) -> dict:
    return {
        "id": c["id"],
        "type": c["type"],
        "type_title": ASSESSMENT_TYPES[c["type"]]["title"],
        "child_name": c.get("child_name", ""),
        "parent_a_name": c.get("parent_a_name", "Parent A"),
        "parent_b_name": c.get("parent_b_name", "Parent B"),
        "has_a": bool(c.get("responses", {}).get("a")),
        "has_b": bool(c.get("responses", {}).get("b")),
        "is_demo": c.get("is_demo", False),
        "created_at": c["created_at"],
    }


class CaseIn(BaseModel):
    type: str
    child_name: str = ""
    parent_a_name: str = "Parent A"
    parent_b_name: str = "Parent B"


class ResponsesIn(BaseModel):
    answers: dict
    primary_concern: str = ""
    goals: list[str] = []


@api.post("/cases")
async def create_case(body: CaseIn, user=Depends(get_current_user)):
    if body.type not in ASSESSMENT_TYPES:
        raise HTTPException(status_code=400, detail="Unknown assessment type")
    cid = str(uuid.uuid4())
    doc = {
        "id": cid,
        "user_id": user["_id"],
        "type": body.type,
        "child_name": body.child_name.strip(),
        "parent_a_name": body.parent_a_name.strip() or "Parent A",
        "parent_b_name": body.parent_b_name.strip() or "Parent B",
        "responses": {},
        "is_demo": False,
        "created_at": datetime.now(timezone.utc).isoformat(),
    }
    await db.cases.insert_one(doc)
    return case_public(doc)


@api.get("/cases")
async def list_cases(user=Depends(get_current_user)):
    cases = await db.cases.find({"user_id": user["_id"]}, {"_id": 0}).sort("created_at", -1).to_list(200)
    return [case_public(c) for c in cases]


async def get_owned_case(case_id: str, user: dict) -> dict:
    c = await db.cases.find_one({"id": case_id, "user_id": user["_id"]}, {"_id": 0})
    if not c:
        raise HTTPException(status_code=404, detail="Assessment not found")
    return c


@api.get("/cases/{case_id}")
async def get_case(case_id: str, user=Depends(get_current_user)):
    return case_public(await get_owned_case(case_id, user))


@api.put("/cases/{case_id}/responses/{parent}")
async def save_responses(case_id: str, parent: str, body: ResponsesIn, user=Depends(get_current_user)):
    if parent not in ("a", "b"):
        raise HTTPException(status_code=400, detail="Parent must be 'a' or 'b'")
    c = await get_owned_case(case_id, user)
    cfg = ASSESSMENT_TYPES[c["type"]]
    valid_keys = {t["key"] for t in cfg["topics"]}
    answers = {k: v for k, v in body.answers.items() if k in valid_keys and v in SCALE}
    if set(answers.keys()) != valid_keys:
        raise HTTPException(status_code=400, detail="All questions must be answered")
    payload = {
        "answers": answers,
        "primary_concern": body.primary_concern.strip(),
        "goals": body.goals,
        "completed_at": datetime.now(timezone.utc).isoformat(),
    }
    await db.cases.update_one({"id": case_id}, {"$set": {f"responses.{parent}": payload, "is_demo": False}})
    c = await get_owned_case(case_id, user)
    return case_public(c)


@api.post("/cases/{case_id}/demo")
async def fill_demo(case_id: str, user=Depends(get_current_user)):
    c = await get_owned_case(case_id, user)
    demo = demo_payload(c["type"])
    await db.cases.update_one(
        {"id": case_id},
        {
            "$set": {
                "responses.a": {**demo["a"], "completed_at": datetime.now(timezone.utc).isoformat()},
                "responses.b": {**demo["b"], "completed_at": datetime.now(timezone.utc).isoformat()},
                "child_name": c.get("child_name") or demo["child_name"],
                "is_demo": True,
            }
        },
    )
    return case_public(await get_owned_case(case_id, user))


@api.get("/cases/{case_id}/report")
async def get_report(case_id: str, user=Depends(get_current_user)):
    c = await get_owned_case(case_id, user)
    if not (c.get("responses", {}).get("a") and c.get("responses", {}).get("b")):
        raise HTTPException(status_code=400, detail="Both parents must complete the assessment before the report can be generated")
    return generate_report(c)


@api.delete("/cases/{case_id}")
async def delete_case(case_id: str, user=Depends(get_current_user)):
    result = await db.cases.delete_one({"id": case_id, "user_id": user["_id"]})
    if result.deleted_count == 0:
        raise HTTPException(status_code=404, detail="Assessment not found")
    return {"ok": True}


async def seed_admin():
    admin_email = os.environ.get("ADMIN_EMAIL")
    admin_password = os.environ.get("ADMIN_PASSWORD")
    if not admin_email or not admin_password:
        return
    existing = await db.users.find_one({"email": admin_email.lower()})
    if existing is None:
        await db.users.insert_one({
            "_id": str(uuid.uuid4()),
            "email": admin_email.lower(),
            "name": "Admin",
            "role": "admin",
            "password_hash": hash_password(admin_password),
            "created_at": datetime.now(timezone.utc).isoformat(),
        })
    elif not verify_password(admin_password, existing["password_hash"]):
        await db.users.update_one(
            {"email": admin_email.lower()},
            {"$set": {"password_hash": hash_password(admin_password)}},
        )


@app.on_event("startup")
async def startup():
    await db.users.create_index("email", unique=True)
    await db.login_attempts.create_index("identifier")
    await db.cases.create_index("user_id")
    await seed_admin()


app.include_router(api)

app.add_middleware(

    allow_credentials=True,
    allow_origins=os.environ.get("CORS_ORIGINS", "*").split(","),
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.on_event("shutdown")
async def shutdown_db_client():
    client.close()

def compute_status(a, b):
    if not a or not b or a == UNSURE or b == UNSURE:
        return "info"
    d = abs(SCALE.index(a) - SCALE.index(b))
    if d == 0:
        return "aligned"
    if d == 1:
        return "close"
    return "different"


def generate_report(case):
    cfg = ASSESSMENT_TYPES[case["type"]]
    ra = case["responses"]["a"]
    rb = case["responses"]["b"]

    topic_results = []
    for t in cfg["topics"]:
        av = ra["answers"].get(t["key"])
        bv = rb["answers"].get(t["key"])
        topic_results.append({
            "key": t["key"],
            "label": t["label"],
            "aligned_text": t["aligned_text"],
            "action_phrase": t["action_phrase"],
            "child_need": t["child_need"],
            "parent_a": av or UNSURE,
            "parent_b": bv or UNSURE,
            "status": compute_status(av, bv),
        })

    aligned = [t for t in topic_results if t["status"] == "aligned"]
    close = [t for t in topic_results if t["status"] == "close"]
    different = [t for t in topic_results if t["status"] == "different"]
    info = [t for t in topic_results if t["status"] == "info"]
    conversation = close + different

    snapshot = {
        "total": len(topic_results),
        "aligned": len(aligned),
        "different": len(conversation),
        "info": len(info),
    }

    focus_source = different[0] if different else (close[0] if close else (info[0] if info else None))
    focus = focus_source["label"].lower() if focus_source else "maintaining the child's current stability"
    priority = cfg["priority_statement"].replace("{focus}", focus)

    questions = []
    for t in conversation[:4]:
        questions.append(
            f"What information would help both parents determine whether {t['action_phrase']} would benefit the child?"
        )
    if not questions:
        questions.append(cfg["central_question"])
    if len(questions) < 2:
        questions.append(cfg["central_question"])

    plan_issues = conversation + info
    action_plan = []
    for t in plan_issues[:3]:
        action_plan.append({
            "issue": t["label"],
            "child_need": t["child_need"],
            "already_agree": aligned[0]["aligned_text"] if aligned else "",
        })

    shared_goals = ra.get("goals") or rb.get("goals") or []

    summary = {
        "aligned": [t["aligned_text"] for t in aligned][:5],
        "info": [f"Additional information about {t['label'].lower()} may help both parents." for t in info][:4],
        "conversation": [f"Parents view {t['label'].lower()} differently and may benefit from a structured conversation." for t in conversation][:4],
        "options": cfg["options"][:6],
        "priority": priority,
    }

    is_court = case["type"] == "court"
    court_map = None
    if is_court:
        court_map = {
            "agreed": [t["label"] for t in aligned],
            "close": [t["label"] for t in close],
            "discussion": [t["label"] for t in different],
            "info": [t["label"] for t in info],
        }

    return {
        "assessment_type": case["type"],
        "assessment_title": cfg["title"],
        "report_title": cfg["report_title"],
        "subtitle": f"Assessment: {cfg['subtitle']}",
        "child_name": case.get("child_name", ""),
        "parent_a_name": case.get("parent_a_name", "Parent A"),
        "parent_b_name": case.get("parent_b_name", "Parent B"),
        "parent_a_status": "Assessment completed",
        "parent_b_status": "Assessment completed",
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "is_demo": case.get("is_demo", False),
        "page_titles": cfg["page_titles"],
        "disclaimer": cfg["disclaimer"],
        "purpose_items": PURPOSE_ITEMS,
        "snapshot": snapshot,
        "snapshot_note": "These numbers are not scores. They are a roadmap for conversation.",
        "shared_goals": shared_goals,
        "parent_a_concern": ra.get("primary_concern", ""),
        "parent_b_concern": rb.get("primary_concern", ""),
        "central_question": cfg["central_question"],
        "alignments": [{"topic": t["label"], "text": t["aligned_text"]} for t in aligned],
        "alignment_foundation_note": ALIGNMENT_FOUNDATION_NOTE,
        "differences": [{"topic": t["label"], "parent_a": t["parent_a"], "parent_b": t["parent_b"]} for t in conversation],
        "differences_note": DIFFERENCES_NOTE,
        "child_needs": cfg["needs"],
        "options": cfg["options"],
        "options_note": "These are not recommendations. They are options for parents to explore.",
        "conversation_questions": questions,
        "professional_map": {

#!/usr/bin/env python3
"""Input a complete joint assessment (Parent A + Parent B) via the SA CoParents API.

Usage:
    python3 input_assessment.py                          # interactive mode (prompts for every answer)
    python3 input_assessment.py --demo educational       # instant demo data for a type
    python3 input_assessment.py --api http://localhost:8001

Assessment types: general, communication, high_conflict, ipv, medical,
                  educational, psychological, court
"""

import argparse
import getpass
import sys

import requests

TYPES = ["general", "communication", "high_conflict", "ipv", "medical", "educational", "psychological", "court"]


def ask(prompt, default=""):
    val = input(f"{prompt}" + (f" [{default}]" if default else "") + ": ").strip()
    return val or default


def ask_choice(question, options):
    print(f"\n{question}")
    for i, o in enumerate(options, 1):
        print(f"  {i}. {o}")
    while True:
        raw = input("  > ").strip()
        if raw.isdigit() and 1 <= int(raw) <= len(options):
            return options[int(raw) - 1]
        print("  Please enter a number from the list.")


def ask_multi(prompt, options):
    print(f"\n{prompt} (comma-separated numbers)")
    for i, o in enumerate(options, 1):
        print(f"  {i}. {o}")
    raw = input("  > ").strip()
    picks = []
    for part in raw.split(","):
        part = part.strip()
        if part.isdigit() and 1 <= int(part) <= len(options):
            picks.append(options[int(part) - 1])
    return picks


def collect_responses(session, api, label, cfg):
    print(f"\n{'=' * 60}\n{label}'s responses\n{'=' * 60}")
    answers = {}
    for idx, t in enumerate(cfg["topics"], 1):
        answers[t["key"]] = ask_choice(f"{idx}. {t['question']}", cfg["scale"])
    concern = input(f"\n{label}'s primary concern (one or two sentences): ").strip()
    goals = ask_multi("Shared goals for the child", cfg["shared_goal_options"])
    return {"answers": answers, "primary_concern": concern, "goals": goals}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--api", default="http://localhost:8001", help="Backend base URL (no /api suffix)")
    parser.add_argument("--demo", choices=TYPES, help="Create a case and fill it with demo data for this type")
    args = parser.parse_args()
    api = args.api.rstrip("/") + "/api"

    session = requests.Session()

    email = ask("Email")
    password = getpass.getpass("Password: ")
    r = session.post(f"{api}/auth/login", json={"email": email, "password": password})
    if r.status_code != 200:
        print(f"Login failed: {r.json().get('detail', r.text)}")
        sys.exit(1)
    print(f"Logged in as {r.json()['email']}")
    token = session.cookies.get("access_token")
    if token:
        session.headers["Authorization"] = f"Bearer {token}"

    if args.demo:
        type_key = args.demo
    else:
        type_key = ask_choice("Choose an assessment type:", TYPES)

    if args.demo:
        case = session.post(f"{api}/cases", json={"type": type_key, "child_name": "Alex",
                                                  "parent_a_name": "Parent A", "parent_b_name": "Parent B"}).json()
        session.post(f"{api}/cases/{case['id']}/demo")
        print(f"\nDemo '{type_key}' assessment created: {case['id']}")
    else:
        cfg = session.get(f"{api}/assessment-types/{type_key}").json()
        child = ask("Child's first name (optional)")
        pa = ask("Parent A name or label", "Parent A")
        pb = ask("Parent B name or label", "Parent B")
        case = session.post(f"{api}/cases", json={"type": type_key, "child_name": child,
                                                  "parent_a_name": pa, "parent_b_name": pb}).json()
        for parent, label in (("a", pa), ("b", pb)):
            payload = collect_responses(session, api, label, cfg)
            r = session.put(f"{api}/cases/{case['id']}/responses/{parent}", json=payload)
            if r.status_code != 200:
                print(f"Failed to save {label}'s responses: {r.json().get('detail', r.text)}")
                sys.exit(1)
        print(f"\nAssessment complete: {case['id']}")

    report = session.get(f"{api}/cases/{case['id']}/report").json()
    s = report["snapshot"]
    print(f"\nReport: {report['report_title']} — {report['subtitle']}")
    print(f"Snapshot: {s['aligned']} aligned | {s['different']} different perspectives | {s['info']} need more info (of {s['total']} topics)")
    print(f"View at: /report/{case['id']}")


if __name__ == "__main__":
    main()

import { createContext, useContext, useEffect, useState } from "react";
import api from "@/lib/api";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.get("/auth/me")
      .then((r) => setUser(r.data))
      .catch(() => setUser(null))
      .finally(() => setLoading(false));
  }, []);

  const login = async (email, password) => {
    const { data } = await api.post("/auth/login", { email, password });
    setUser(data);
    return data;
  };

  const register = async (payload) => {
    const { data } = await api.post("/auth/register", payload);
    setUser(data);
    return data;
  };

  const logout = async () => {
    try {
      await api.post("/auth/logout");
    } finally {
      setUser(null);
    }
  };

  return (
    <AuthContext.Provider value={{ user, loading, login, register, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

"use client";
// Inspired by react-hot-toast library
import * as React from "react"

const TOAST_LIMIT = 1
const TOAST_REMOVE_DELAY = 1000000

const actionTypes = {
  ADD_TOAST: "ADD_TOAST",
  UPDATE_TOAST: "UPDATE_TOAST",
  DISMISS_TOAST: "DISMISS_TOAST",
  REMOVE_TOAST: "REMOVE_TOAST"
}

let count = 0

function genId() {
  count = (count + 1) % Number.MAX_SAFE_INTEGER
  return count.toString();
}

const toastTimeouts = new Map()

const addToRemoveQueue = (toastId) => {
  if (toastTimeouts.has(toastId)) {
    return
  }

  const timeout = setTimeout(() => {
    toastTimeouts.delete(toastId)
    dispatch({
      type: "REMOVE_TOAST",
      toastId: toastId,
    })
  }, TOAST_REMOVE_DELAY)

  toastTimeouts.set(toastId, timeout)
}

export const reducer = (state, action) => {
  switch (action.type) {
    case "ADD_TOAST":
      return {
        ...state,
        toasts: [action.toast, ...state.toasts].slice(0, TOAST_LIMIT),
      };

    case "UPDATE_TOAST":
      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === action.toast.id ? { ...t, ...action.toast } : t),
      };

    case "DISMISS_TOAST": {
      const { toastId } = action

      // ! Side effects ! - This could be extracted into a dismissToast() action,
      // but I'll keep it here for simplicity
      if (toastId) {
        addToRemoveQueue(toastId)
      } else {
        state.toasts.forEach((toast) => {
          addToRemoveQueue(toast.id)
        })
      }

      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === toastId || toastId === undefined
            ? {
                ...t,
                open: false,
              }
            : t),
      };
    }
    case "REMOVE_TOAST":
      if (action.toastId === undefined) {
        return {
          ...state,
          toasts: [],
        }
      }
      return {
        ...state,
        toasts: state.toasts.filter((t) => t.id !== action.toastId),
      };
  }
}

const listeners = []

let memoryState = { toasts: [] }

function dispatch(action) {
  memoryState = reducer(memoryState, action)
  listeners.forEach((listener) => {
    listener(memoryState)
  })
}

function toast({
  ...props
}) {
  const id = genId()

  const update = (props) =>
    dispatch({
      type: "UPDATE_TOAST",
      toast: { ...props, id },
    })
  const dismiss = () => dispatch({ type: "DISMISS_TOAST", toastId: id })

  dispatch({
    type: "ADD_TOAST",
    toast: {
      ...props,
      id,
      open: true,
      onOpenChange: (open) => {
        if (!open) dismiss()
      },
    },
  })

  return {
    id: id,
    dismiss,
    update,
  }
}

function useToast() {
  const [state, setState] = React.useState(memoryState)

  React.useEffect(() => {
    listeners.push(setState)
    return () => {
      const index = listeners.indexOf(setState)
      if (index > -1) {
        listeners.splice(index, 1)
      }
    };
  }, [state])

  return {
    ...state,
    toast,
    dismiss: (toastId) => dispatch({ type: "DISMISS_TOAST", toastId }),
  };
}

export default function Wizard() {
  const [types, setTypes] = useState([]);
  const [step, setStep] = useState(0);
  const [typeKey, setTypeKey] = useState(null);
  const [config, setConfig] = useState(null);
  const [caseId, setCaseId] = useState(null);
  const [names, setNames] = useState({ child_name: "", parent_a_name: "", parent_b_name: "" });
  const [busy, setBusy] = useState(false);
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();

  useEffect(() => {
    api.get("/assessment-types").then((r) => setTypes(r.data)).catch((e) => toast.error(formatApiError(e)));
  }, []);

  const resumeCase = async (id) => {
    try {
      const { data: c } = await api.get(`/cases/${id}`);
      const { data: cfg } = await api.get(`/assessment-types/${c.type}`);
      setTypeKey(c.type);
      setConfig(cfg);
      setCaseId(c.id);
      setNames({ child_name: c.child_name, parent_a_name: c.parent_a_name, parent_b_name: c.parent_b_name });
      setStep(!c.has_a ? 2 : 3);
    } catch (e) {
      toast.error(formatApiError(e));
    }
  };

  useEffect(() => {
    const existing = searchParams.get("case");
    if (existing) resumeCase(existing);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const selectType = async (key) => {
    setTypeKey(key);
    try {
      const { data } = await api.get(`/assessment-types/${key}`);
      setConfig(data);
      setStep(1);
    } catch (e) {
      toast.error(formatApiError(e));
    }
  };

  const createCase = async () => {
    setBusy(true);
    try {
      const { data } = await api.post("/cases", { type: typeKey, ...names });
      setCaseId(data.id);
      setStep(2);
    } catch (e) {
      toast.error(formatApiError(e));
    } finally {
      setBusy(false);
    }
  };

  const submitResponses = async (parent, payload) => {
    setBusy(true);
    try {
      const { data } = await api.put(`/cases/${caseId}/responses/${parent}`, payload);
      if (parent === "a") {
        setStep(3);
      } else {
        navigate(`/report/${caseId}`);
      }
      return data;
    } catch (e) {
      toast.error(formatApiError(e));
    } finally {
      setBusy(false);
    }
  };

  const fillDemo = async () => {
    setBusy(true);
    try {
      await api.post(`/cases/${caseId}/demo`);
      navigate(`/report/${caseId}`);
    } catch (e) {
      toast.error(formatApiError(e));
    } finally {
      setBusy(false);
    }
  };

  const steps = ["Assessment type", "Details", `${names.parent_a_name || "Parent A"}`, `${names.parent_b_name || "Parent B"}`];

  return (
    <div className="min-h-screen bg-background" data-testid="wizard-page">
      <header className="bg-white border-b border-border no-print">
        <div className="max-w-3xl mx-auto px-6 py-4 flex items-center justify-between">
          <Link to="/dashboard" className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground" data-testid="wizard-back-link">
            <ArrowLeft className="h-4 w-4" /> Dashboard
          </Link>
          <div className="flex items-center gap-2">
            <FileText className="h-5 w-5 text-primary" />
            <span className="font-display text-xl font-semibold text-primary">SA CoParents</span>
          </div>
        </div>
      </header>

      <main className="max-w-3xl mx-auto px-6 py-10">
        <div className="flex items-center gap-2 mb-10" data-testid="wizard-progress">
          {steps.map((s, i) => (
            <div key={s} className="flex items-center gap-2 flex-1 last:flex-none">
              <span
                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-semibold ${i <= step ? "bg-primary text-primary-foreground" : "bg-muted text-muted-foreground"}`}
                data-testid={`step-indicator-${i}`}
              >
                {i + 1}
              </span>
              <span className={`text-xs hidden sm:block ${i === step ? "font-semibold" : "text-muted-foreground"}`}>{s}</span>
              {i < steps.length - 1 && <div className="flex-1 h-px bg-border mx-2" />}
            </div>
          ))}
        </div>

        {step === 0 && (
          <div data-testid="step-type">
            <h1 className="text-4xl font-semibold mb-2">Choose an assessment</h1>
            <p className="text-muted-foreground mb-8">Both parents will answer the same questions. The report combines both perspectives.</p>
            <div className="grid sm:grid-cols-2 gap-4">
              {types.map((t) => (
                <button
                  key={t.key}
                  onClick={() => selectType(t.key)}
                  className="text-left bg-white border border-border rounded-lg p-6 hover:-translate-y-0.5 hover:shadow-md hover:border-primary transition-[transform,box-shadow,border-color] duration-200"
                  data-testid={`type-option-${t.key}`}
                >
                  <p className="font-semibold">{t.title}</p>
                  <p className="text-sm text-muted-foreground mt-2 leading-relaxed">{t.description}</p>
                  <p className="text-xs text-accent font-medium mt-3">{t.topic_count} topics</p>
                </button>
              ))}
            </div>
          </div>
        )}

        {step === 1 && (

            <h1 className="text-4xl font-semibold mb-2">Who is this assessment for?</h1>
            <p className="text-muted-foreground mb-8">These names appear on the report. Initials or first names are fine.</p>

              </div>
              <div className="space-y-2">
                <Label htmlFor="parent-a">Parent A name or label</Label>
                <Input id="parent-a" className="h-12" value={names.parent_a_name} onChange={(e) => setNames({ ...names, parent_a_name: e.target.value })} placeholder="Parent A" data-testid="parent-a-name-input" />
              </div>
              <div className="space-y-2">
                <Label htmlFor="parent-b">Parent B name or label</Label>
                <Input id="parent-b" className="h-12" value={names.parent_b_name} onChange={(e) => setNames({ ...names, parent_b_name: e.target.value })} placeholder="Parent B" data-testid="parent-b-name-input" />
              </div>
              <div className="pt-2 flex justify-between items-center">

                </button>

            </div>
          </div>
        )}

              <p className="text-sm">Exploring the platform? Load realistic sample data instead of answering manually.</p>

              </Button>

              parentLabel={names.parent_a_name || "Parent A"}
              busy={busy}
              onSubmit={(payload) => submitResponses("a", payload)}
            />
          </div>
        )}

        {step === 3 && config && (
          <div data-testid="step-parent-b">

  );

# Joint Alignment Report — Integration Guide

This document is written for the agent working in the **sacoparentsapp.online** Emergent project.
It describes how to port the Joint Assessment / Joint Alignment Report module from this repository
into that app. The module is intentionally self-contained.

## What This Module Is

A universal 10-page **Joint Alignment Report** system: two parents each complete the same structured
assessment (10 topics, 5-point concern scale), and a rule-based engine generates a 10-page report
(alignment, differences in perspective, child needs, options, conversation questions, professional
discussion map, action plan worksheet, alignment summary). Eight assessment types: General
Co-Parenting, Communication, High-Conflict, IPV/Family Violence (safety-first variant), Medical,
Educational, Psychological/Behavioral, Preparing for Court.

## Files to Port (copy these)

### Backend — copy verbatim, no changes needed
| File | Contents |
|---|---|
| `backend/assessment_data.py` | All 8 assessment type configs: topics, questions, scales, options, child needs, page titles, disclaimers, demo payloads. No dependencies. |
| `backend/report_engine.py` | `generate_report(case)` — pure rule-based function, takes a case dict, returns the full report JSON. Imports only from `assessment_data`. |

### Backend — port the routes, adapt auth
From `backend/server.py`, copy ONLY these route groups into the target app's router:
- `GET  /api/assessment-types` and `GET /api/assessment-types/{key}` — static config, no auth changes needed.
- `POST /api/cases`, `GET /api/cases`, `GET /api/cases/{id}`, `PUT /api/cases/{id}/responses/{a|b}`, `POST /api/cases/{id}/demo`, `GET /api/cases/{id}/report`, `DELETE /api/cases/{id}` — replace `get_current_user` with the target app's existing auth dependency, and `user["_id"]` with the target app's user id field. Cases are stored in a `cases` MongoDB collection with a string `id` (uuid), `user_id`, `type`, `child_name`, `parent_a_name`, `parent_b_name`, `responses: {a, b}`, `is_demo`, `created_at`.

**DO NOT copy** the auth routes (`/api/auth/*`), JWT helpers, bcrypt code, admin seeding, or CORS setup — the target app already has its own auth. Reuse it.

### Frontend — port and restyle
| File | Role |
|---|---|
| `frontend/src/pages/Wizard.jsx` | 4-step intake: choose type → case details → Parent A form → Parent B form. Includes `ParentForm` component (radio scale per topic, primary concern textarea, shared-goals checkboxes) and demo-data buttons. |
| `frontend/src/pages/ReportView.jsx` | Report viewer shell: page navigation (dots/prev/next), print button, print-only container rendering all 10 pages. |
| `frontend/src/components/report/Pages.jsx` | The 10 report page components (`REPORT_PAGES` array). Consumes the report JSON directly. |
| `frontend/src/components/report/Paper.jsx` | Paper shell, `Snapshot` donut chart (uses recharts), `PullQuote`, `StatusDot`, `SectionHeading`. |

**DO NOT copy** `Landing.jsx`, `Auth.jsx`, `AuthContext.jsx`, or `lib/api.js` verbatim — adapt API calls to the target app's existing API client/auth context. Skip this app's auth pages entirely.

## Frontend Dependencies to Check
- `recharts` (snapshot donut) — `yarn add recharts` if missing.
- `lucide-react`, `sonner`, shadcn `button/input/label/textarea/checkbox/radio-group` — already standard in Emergent templates.

## Styles to Port
From `frontend/src/index.css`: the `.report-paper`, `.report-page-enter`, `@keyframes pageFade`, `.worksheet-line`, `.print-only`, and the `@media print` block (critical for print/PDF output).

## Report JSON Contract
`GET /api/cases/{id}/report` returns everything `Pages.jsx` needs: `page_titles[10]`, `disclaimer`, `snapshot {total, aligned, different, info}`, `shared_goals[]`, `parent_a_concern`, `parent_b_concern`, `central_question`, `alignments[{topic, text}]`, `differences[{topic, parent_a, parent_b}]`, `child_needs[{title, text}]`, `options[]`, `conversation_questions[]`, `professional_map`, `action_plan[{issue, child_need, already_agree}]`, `summary {aligned, info, conversation, options, priority}`, `court_map` (court type only), `philosophy`, `is_demo`. If you change the backend shape, update the page components to match.

## Content Rules (important — preserve these when editing)
- Differences are "differences in perspective", never conflicts/failures/discrepancies.
- Options are "options for parents to explore", never recommendations.
- Psychological report: describe parental observations, never diagnose.
- IPV report: safety first — "Agreement between parents is not the measure of success where safety is a concern."
- Every report ends with: "We value the parenting system by how well it supports the child, not by how friendly the adults appear."

3. Restyle to the target app's design system.
4. Test end-to-end: create case → demo data → all 10 pages → print.
